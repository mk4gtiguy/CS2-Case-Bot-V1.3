ACTION: CS2_OpenCase

Called by Case_Command. Do not link directly to a chat command.

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        var random = new Random();
        int roll = random.Next(1, 1001);

        string rarityType;
        if (roll <= 26) rarityType = "Gold";
        else if (roll >= 975) rarityType = "Red";
        else if (roll >= 950) rarityType = "Pink";
        else if (roll >= 900) rarityType = "Purple";
        else rarityType = "Blue";

        string emoji;
        double value;
        if (rarityType == "Gold") { emoji = "🟡"; value = 50.00; }
        else if (rarityType == "Red") { emoji = "🔴"; value = 5.00; }
        else if (rarityType == "Pink") { emoji = "💗"; value = 1.50; }
        else if (rarityType == "Purple") { emoji = "🟣"; value = 0.50; }
        else { emoji = "🔷"; value = 0.10; }

        string user = CPH.GetGlobalVar<string>("User");
        string currentCase = CPH.GetGlobalVar<string>("CurrentCase");
        string casePrefix = currentCase.Replace(" ", "").Replace("&", "").Replace("-", "");

        string skinList = CPH.GetGlobalVar<string>(casePrefix + rarityType + "List");
        if (string.IsNullOrEmpty(skinList))
            skinList = CPH.GetGlobalVar<string>(rarityType + "List");

        if (string.IsNullOrEmpty(skinList))
        {
            CPH.SendMessage(user + " error: no skin list found for " + currentCase + "!");
            return false;
        }

        var validSkins = new System.Collections.Generic.List<string>();
        foreach (string skin in skinList.Split(';'))
            if (!string.IsNullOrWhiteSpace(skin))
                validSkins.Add(skin.Trim());

        if (validSkins.Count == 0)
        {
            CPH.SendMessage(user + " error: " + currentCase + " has no skins configured!");
            return false;
        }

        string currentSkin = validSkins[random.Next(0, validSkins.Count)];
        string newItem = emoji + " " + currentSkin;

        // Save to inventory
        string oldItems = CPH.GetGlobalVar<string>(user + "_Items");
        string newItemsList = string.IsNullOrEmpty(oldItems) ? newItem : oldItems + ";" + newItem;
        CPH.SetGlobalVar(user + "_Items", newItemsList, true);
        CPH.SetGlobalVar(user + "_LastCase", currentCase, true);

        // Track case opens per case
        string caseKey = casePrefix + "_Opened";
        int caseOpened = CPH.GetGlobalVar<int>(caseKey);
        CPH.SetGlobalVar(caseKey, caseOpened + 1, true);

        // Track per user per case
        int userCaseOpened = CPH.GetGlobalVar<int>(user + "_" + casePrefix + "_Opened");
        CPH.SetGlobalVar(user + "_" + casePrefix + "_Opened", userCaseOpened + 1, true);

        // Track total opens
        int totalOpened = CPH.GetGlobalVar<int>(user + "_TotalOpened");
        CPH.SetGlobalVar(user + "_TotalOpened", totalOpened + 1, true);

        // Track rarity counts per user
        string rarityCounterKey = user + "_" + rarityType + "s";
        int rarityCount = CPH.GetGlobalVar<int>(rarityCounterKey);
        CPH.SetGlobalVar(rarityCounterKey, rarityCount + 1, true);

        // Global gold tracking
        if (rarityType == "Gold")
        {
            int userGolds = CPH.GetGlobalVar<int>(user + "_Golds");
            CPH.SetGlobalVar(user + "_Golds", userGolds + 1, true);
            int totalGolds = CPH.GetGlobalVar<int>("GoldPulls_Total");
            CPH.SetGlobalVar("GoldPulls_Total", totalGolds + 1, true);
        }

        // Track all users for leaderboards
        string allUsers = CPH.GetGlobalVar<string>("AllUsers");
        if (string.IsNullOrEmpty(allUsers))
        {
            allUsers = user;
        }
        else
        {
            bool userExists = false;
            foreach (string existingUser in allUsers.Split('§'))
                if (existingUser.Equals(user, StringComparison.OrdinalIgnoreCase))
                { userExists = true; break; }
            if (!userExists) allUsers += "§" + user;
        }
        CPH.SetGlobalVar("AllUsers", allUsers, true);

        CPH.SetGlobalVar("CurrentSkin", currentSkin, true);
        CPH.SetGlobalVar("RarityType", rarityType, true);
        CPH.SetGlobalVar("Emoji", emoji, true);
        CPH.SetGlobalVar("Value", value.ToString("F2"), true);
        CPH.SetGlobalVar("CurrentValue", value, true);

        CPH.SendMessage(user + " opened " + currentCase + " and got: " + emoji + " " + currentSkin + " (Value: $" + value.ToString("F2") + ")");
        return true;
    }
}
```

ACTION: SetCase_Command (!setcase)

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        string user = args["userName"].ToString();
        string rawInput = args["rawInput"].ToString();
        
        // Remove "!setcase " from the input
        string caseName = rawInput.Replace("!setcase", "").Trim();
        
        // Valid cases list
        string[] validCases = { "Dreams", "Recoil", "Kilowatt", "Revolution", "Fracture", "Clutch" };
        bool isValid = false;
        
        foreach (string validCase in validCases)
        {
            if (caseName.Equals(validCase, StringComparison.OrdinalIgnoreCase))
            {
                caseName = validCase;
                isValid = true;
                break;
            }
        }
        
        if (!isValid)
        {
            CPH.SendMessage("📦 " + user + " Unknown case: '" + caseName + "' | Usage: !setcase [CaseName] | Available: Dreams, Recoil, Kilowatt, Revolution, Fracture, Clutch | Example: !setcase Dreams");
            return false;
        }
        
        CPH.SetGlobalVar(user + "_LastCase", caseName, true);
        
        // Get cost for the selected case
        double cost = 1.50;
        if (caseName.Equals("Recoil", StringComparison.OrdinalIgnoreCase)) cost = 0.50;
        else if (caseName.Equals("Fracture", StringComparison.OrdinalIgnoreCase) || 
                 caseName.Equals("Kilowatt", StringComparison.OrdinalIgnoreCase)) cost = 2.00;
        else if (caseName.Equals("Clutch", StringComparison.OrdinalIgnoreCase)) cost = 3.00;
        
        CPH.SendMessage("📦 " + user + " switched to " + caseName + " case! ($" + cost.ToString("F2") + " per open) | Type !case to open it, or !setcase [name] to switch again.");
        return true;
    }
}
```

ACTION: Case_Command (!case)

Must have Sub-Action: Run Action → CS2_OpenCase

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        string user = args["userName"].ToString();
        string currentCase = CPH.GetGlobalVar<string>(user + "_LastCase");
        
        if (string.IsNullOrEmpty(currentCase))
        {
            CPH.SendMessage(user + " please set a case first with !setcase Dreams (or Recoil, Kilowatt, etc.)");
            return false;
        }
        
        CPH.SetGlobalVar("User", user, true);
        CPH.SetGlobalVar("CurrentCase", currentCase, true);
        
        // Get case prefix for cost calculation
        string casePrefix = currentCase.Replace(" ", "").Replace("&", "").Replace("-", "");
        
        // Determine cost
        double cost = 1.50;
        if (casePrefix.Equals("Recoil", StringComparison.OrdinalIgnoreCase)) cost = 0.50;
        else if (casePrefix.Equals("Fracture", StringComparison.OrdinalIgnoreCase) || 
                 casePrefix.Equals("Kilowatt", StringComparison.OrdinalIgnoreCase)) cost = 2.00;
        else if (casePrefix.Equals("Clutch", StringComparison.OrdinalIgnoreCase)) cost = 3.00;
        
        // Check balance
        double balance = CPH.GetGlobalVar<double>(user + "_Balance");
        
        if (balance < cost)
        {
            double shortfall = cost - balance;
            CPH.SendMessage(user + " you need $" + cost.ToString("F2") + " to open " + currentCase + " but only have $" + balance.ToString("F2") + " — you're $" + shortfall.ToString("F2") + " short! Try !daily or !sell.");
            return false;
        }
        
        // Deduct cost
        balance -= cost;
        CPH.SetGlobalVar(user + "_Balance", balance, true);
        CPH.SetGlobalVar("CaseCost", cost, true);
        
        // Show purchase confirmation
        CPH.SendMessage("🔑 " + user + " bought a " + currentCase + " key for $" + cost.ToString("F2") + " | Balance: $" + balance.ToString("F2"));
        
        return true;
    }
}
```

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        string user = args["userName"].ToString();
        string lastDailyStr = CPH.GetGlobalVar<string>(user + "_LastDaily");

        if (!string.IsNullOrEmpty(lastDailyStr))
        {
            DateTime lastDaily = DateTime.Parse(lastDailyStr);
            TimeSpan elapsed = DateTime.UtcNow - lastDaily;
            if (elapsed.TotalHours < 24)
            {

                CPH.SendMessage(user + " daily already claimed! " + hours + "h remaining.");

        var rng = new Random();
        double reward = rng.Next(500, 2501) / 100.0;
        double balance = CPH.GetGlobalVar<double>(user + "_Balance");
        balance += reward;

        CPH.SetGlobalVar(user + "_Balance", balance, true);
        CPH.SetGlobalVar(user + "_LastDaily", DateTime.UtcNow.ToString(), true);
        CPH.SendMessage("🎁 " + user + " claimed $" + reward.ToString("F2") + " from !daily! | Balance: $" + balance.ToString("F2"));
        return true;
    }
```

```csharp
using System;
using System.Collections.Generic;

public class CPHInline
{
    public bool Execute()
    {
        string user = args["userName"].ToString();
        string items = CPH.GetGlobalVar<string>(user + "_Items");

        if (string.IsNullOrEmpty(items)) 
        { 
            CPH.SendMessage(user + " your inventory is empty!"); 
            return false; 
        }

        string[] itemList = items.Split(';');
        int blueCount = 0;
        
        foreach (string item in itemList)
        {
            if (item.Contains("🔷")) blueCount++;
        }

        if (blueCount < 10)
        {
            CPH.SendMessage(user + " you need 10 🔷 Blue skins to trade up! You have " + blueCount + ". Type !inv to see your inventory.");
            return false;
        }

        // Remove exactly 10 blue items
        var newList = new List<string>(itemList);
        int removed = 0;
        for (int i = newList.Count - 1; i >= 0 && removed < 10; i--)
        {
            if (newList[i].Contains("🔷")) 
            { 
                newList.RemoveAt(i); 
                removed++; 
            }
        }

        // Get current case for purple skin
        string currentCase = CPH.GetGlobalVar<string>("CurrentCase");
        string casePrefix = string.IsNullOrEmpty(currentCase) ? "Dreams" : currentCase.Replace(" ", "").Replace("&", "").Replace("-", "");
        string purpleList = CPH.GetGlobalVar<string>(casePrefix + "PurpleList");
        
        if (string.IsNullOrEmpty(purpleList))
        {
            CPH.SendMessage(user + " error: no purple skins found for " + currentCase + " case!");
            return false;
        }

        string[] purples = purpleList.Split(';');
        var rng = new Random();
        string newSkin = purples[rng.Next(0, purples.Length)].Trim();
        string newItem = "🟣 " + newSkin;
        newList.Add(newItem);

        CPH.SetGlobalVar(user + "_Items", string.Join(";", newList), true);
        CPH.SendMessage("🎉 " + user + " traded up 10 🔷 Blues and got: 🟣 " + newSkin + "!");
        return true;
```

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        string user = args["userName"].ToString();
        string items = CPH.GetGlobalVar<string>(user + "_Items");

        if (string.IsNullOrEmpty(items)) { CPH.SendMessage(user + " your inventory is empty!"); return false; }

        string[] itemList = items.Split(';');
        var rng = new Random();
        int index = rng.Next(0, itemList.Length);
        string contributed = itemList[index].Trim();

        int tickets = 1;
        if (contributed.Contains("🟡")) tickets = 250;
        else if (contributed.Contains("🔴")) tickets = 50;
        else if (contributed.Contains("💗")) tickets = 15;
        else if (contributed.Contains("🟣")) tickets = 5;

        var newList = new System.Collections.Generic.List<string>(itemList);
        newList.RemoveAt(index);
        CPH.SetGlobalVar(user + "_Items", string.Join(";", newList), true);

        string pot = CPH.GetGlobalVar<string>("JackpotPot");
        string entries = CPH.GetGlobalVar<string>("JackpotEntries");

        string newEntry = user + "~" + contributed + "~" + tickets;
        pot = string.IsNullOrEmpty(pot) ? newEntry : pot + "§" + newEntry;

        string currentEntries = string.IsNullOrEmpty(entries) ? "" : entries;
        for (int i = 0; i < tickets; i++)
            currentEntries += (string.IsNullOrEmpty(currentEntries) ? "" : "§") + user;

        CPH.SetGlobalVar("JackpotPot", pot, true);
        CPH.SetGlobalVar("JackpotEntries", currentEntries, true);

        string[] potEntries = pot.Split('§');
        double potValue = 0;
        foreach (string entry in potEntries)
        {
            string[] parts = entry.Split('~');
            if (parts.Length >= 2)
            {
                if (parts[1].Contains("🟡")) potValue += 50.00;
                else if (parts[1].Contains("🔴")) potValue += 5.00;
                else if (parts[1].Contains("💗")) potValue += 1.50;
                else if (parts[1].Contains("🟣")) potValue += 0.50;
                else potValue += 0.10;
            }
        }

        CPH.SendMessage("🎰 " + user + " entered jackpot with " + contributed + " (" + tickets + " tickets) | Pot: $" + potValue.ToString("F2"));
        return true;
    }
}
```

ACTION: JackpotInfo_Command (!jackpotinfo)

```csharp
using System;
using System.Collections.Generic;

public class CPHInline
{
    public bool Execute()
    {
        string pot = CPH.GetGlobalVar<string>("JackpotPot");
        string entries = CPH.GetGlobalVar<string>("JackpotEntries");

        if (string.IsNullOrEmpty(pot))
        {
            CPH.SendMessage("🎰 No active jackpot! Type !jackpot to enter with a random item.");
            return true;
        }

        string[] potEntries = pot.Split(new[] { '§' }, StringSplitOptions.RemoveEmptyEntries);
        var players = new List<string>();
        double potValue = 0;

        foreach (string entry in potEntries)
        {
            string[] parts = entry.Split('~');
            if (parts.Length >= 2)
            {
                string player = parts[0];
                string item = parts[1];
                
                if (!players.Contains(player))
                    players.Add(player);
                
                if (item.Contains("🟡")) potValue += 50.00;
                else if (item.Contains("🔴")) potValue += 5.00;
                else if (item.Contains("💗")) potValue += 1.50;
                else if (item.Contains("🟣")) potValue += 0.50;
                else potValue += 0.10;
            }
        }

        int tickets = 0;
        if (!string.IsNullOrEmpty(entries))
        {
            tickets = entries.Split(new[] { '§' }, StringSplitOptions.RemoveEmptyEntries).Length;
        }
        
        string playerList = string.Join(", ", players);
        CPH.SendMessage("🎰 Current Jackpot — Players: " + players.Count + " (" + playerList + ") | Items: " + potEntries.Length + " | Pot: $" + potValue.ToString("F2") + " | Tickets: " + tickets);
        return true;
    }
}
```

```csharp
using System;

public class CPHInline
{
    public bool Execute()
    {
        string entries = CPH.GetGlobalVar<string>("JackpotEntries");
        string pot = CPH.GetGlobalVar<string>("JackpotPot");

        if (string.IsNullOrEmpty(entries))
        {
            CPH.SendMessage("No jackpot entries yet!");
            return false;
        }

        string[] entryList = entries.Split(new[] { '§' }, StringSplitOptions.RemoveEmptyEntries);
        if (entryList.Length == 0)
        {
            CPH.SendMessage("No valid jackpot entries!");
            return false;
        }

        var rng = new Random();
        string winner = entryList[rng.Next(0, entryList.Length)];

        string[] potEntries = pot.Split(new[] { '§' }, StringSplitOptions.RemoveEmptyEntries);
        var winnings = new System.Collections.Generic.List<string>();
        double potValue = 0;

        foreach (string entry in potEntries)
        {
            string[] parts = entry.Split('~');
            if (parts.Length >= 2)
            {
                winnings.Add(parts[1]);
                if (parts[1].Contains("🟡")) potValue += 50.00;
                else if (parts[1].Contains("🔴")) potValue += 5.00;
                else if (parts[1].Contains("💗")) potValue += 1.50;
                else if (parts[1].Contains("🟣")) potValue += 0.50;
                else potValue += 0.10;
            }
        }

        string winnerItems = CPH.GetGlobalVar<string>(winner + "_Items");
        string allWinnings = string.Join(";", winnings);
        
        if (string.IsNullOrEmpty(winnerItems))
            CPH.SetGlobalVar(winner + "_Items", allWinnings, true);
        else
            CPH.SetGlobalVar(winner + "_Items", winnerItems + ";" + allWinnings, true);

        CPH.SetGlobalVar("JackpotPot", "", true);
        CPH.SetGlobalVar("JackpotEntries", "", true);

        CPH.SendMessage("🎰 JACKPOT WINNER! " + winner + " won " + winnings.Count + " items worth $" + potValue.ToString("F2") + "!");
        return true;
    }
}
```
